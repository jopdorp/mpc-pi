# Raspberry Pi 5 boot configuration for the MPC2000XL appliance.
kernel=Image
arm_64bit=1
disable_overscan=1

# ===== AUDIO =====
# Onboard/HDMI audio is off so the I2S card is the only sound device and
# always lands on card 0; MAME's PipeWire path picks the default sink.
dtparam=audio=off
dtoverlay=vc4-kms-v3d,noaudio

# PCM5102A DAC plus PCM1808 ADC on the shared I2S bus. See
# overlays/mpc-audio-overlay.dts for the wiring this assumes.
dtoverlay=mpc-audio

# ===== DISPLAY =====
# The panel LCD is delivered to external hardware over USB by the MK1 daemon,
# so no display is required in normal operation. HDMI stays available for
# bring-up and is harmless when nothing is attached.
display_auto_detect=1

# ===== SERIAL CONSOLE =====
enable_uart=1

# The Pi 5 sustains 2.4 GHz stock; raise only with real cooling and after
# re-running the throughput measurements, since the emulator is latency
# sensitive and thermal throttling shows up as audio dropouts.
# arm_freq=2800
